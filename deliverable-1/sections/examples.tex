\subsection{Παραδείγματα Ερωτημάτων}

\begin{enumerate}

\item Συνολικός αριθμός BTC κρυπτονομισμάτων χρήστη με user\_id = 1
\[
\begin{aligned}
&A \leftarrow \text{User\_has\_Wallet}
\bowtie_{\text{wallet\_id}} \text{Wallet\_stores\_Cryptocurrency}
\\
&B \leftarrow
{}_{\text{user\_id, ticker\_symbol}}\mathcal{G}_{\text{sum(crypto\_amount) as cryptoAmount}}
\Bigl(\pi_{\text{user\_id, ticker\_symbol, crypto\_amount}} \bigl(A\bigr)\Bigr) \\
&C \leftarrow \sigma_{\text{user\_id=1} \land \text{ticker\_symbol=BTC}}\bigl(B\bigr)
\end{aligned}
\]

\item Όλα τα πορτοφόλια του χρήστη με user\_id = 1 (κάτοχος ή μέλος) που έχουν λάβει μέρος σε τραπεζικές συναλλαγές
\[
\begin{aligned}
&A \leftarrow 
\bigl(
  \text{User\_has\_Wallet} \bowtie_{\text{wallet\_id}}
  \text{Wallet} \bowtie_{\text{wallet\_id}}
  \text{BankAccount\_transfers\_to\_Wallet}
\bigr) \\
&B \leftarrow \sigma_{\text{user\_id=1}} \Bigl(\pi_{\text{user\_id, wallet\_id, Wallet.name}}
\bigl(A\bigr)\Bigr) \\
\end{aligned}
\]

\item Όλες οι προσκλήσεις με παραλήπτη ή αποστολέα που δεν είναι ο χρήστης με user\_id = 1
\[
\text{Invitation} - \sigma_{\text{inviter\_id=1} \lor \text{invitee\_id=1}}\bigl(\text{Invitation}\bigr)
\]

\item Συναλλαγές BTC κρυπτονομισμάτων του χρήστη με user\_id = 1 που πραγματοποιήθηκαν μετά τις
2025-12-17T16:05:42Z
\[
\begin{aligned}
\sigma_{\text{user\_id} = 1 \land \text{ticker\_symbol = BTC} \land 
\text{timestamp > 2025-12-17T16:05:42Z}} \bigl(\text{Transaction}\bigr)
\end{aligned}
\]


\end{enumerate}
