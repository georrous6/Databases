\subsection{Όψεις}

\begin{enumerate}

\item Όψη για τα υπόλοιπα κρυπτονομισμάτων των χρηστών
\[
\begin{aligned}
&\text{A} \leftarrow \text{User\_has\_Wallet}
\bowtie_{\text{wallet\_id}} \text{CryptoBalance}
\bowtie_{\text{ticker\_symbol}} \text{Cryptocurrency}
\\
&\textbf{UserCryptoBalances} =
\pi_{\text{user\_id, name, ticker\_symbol, crypto\_amount}} \bigl(A\bigr)
\end{aligned}
\]

\item Όψη για τις τραπεζικές συναλλαγές των χρηστών
\[
\begin{aligned}
&A \leftarrow 
\bigl(
  \text{User} \bowtie_{\text{user\_id}}
  \text{BankAccount} \bowtie_{\text{account\_id}}
  \text{transfers\_to}
\bigr) \\
&\textbf{UserBankTransactions} = \\
&\pi_{\text{user\_id, name, account\_id, bank\_name, holder\_name, 
encrypted\_iban, wallet\_id, direction, timestamp, amount}}\bigl(A\bigr) \\
\end{aligned}
\]

\item Όψη για τα candlesticks των κρυπτονομισμάτων
\[
\begin{aligned}
& A \leftarrow \text{Cryptocurrency} \bowtie_{\text{ticker\_symbol}} \text{Candlestick} \\
& \textbf{CryptoCandlestick} = \\
& \pi_{\text{ticker\_symbol, name, candle\_id, interval, open\_price,
high\_price, low\_price, close\_price, volume}}
\bigl(A\bigr)
\end{aligned}
\]

\end{enumerate}
